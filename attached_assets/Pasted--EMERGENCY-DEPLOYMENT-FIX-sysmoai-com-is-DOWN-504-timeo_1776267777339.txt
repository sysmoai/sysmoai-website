════════════════════════════════════════
STEP 2: Create Production Server
════════════════════════════════════════

Create a new file called server.js in the PROJECT ROOT 
(same level as package.json, NOT inside src/):

const express = require('express');
const compression = require('compression');
const path = require('path');

const app = express();
const PORT = process.env.PORT || 3000;

// Enable gzip compression for all responses
app.use(compression());

// Security headers
app.use((req, res, next) => {
  res.setHeader('X-Content-Type-Options', 'nosniff');
  res.setHeader('X-Frame-Options', 'DENY');
  res.setHeader('X-XSS-Protection', '1; mode=block');
  res.setHeader('Referrer-Policy', 'strict-origin-when-cross-origin');
  next();
});

// Cache static assets aggressively (JS, CSS, images)
app.use('/assets', express.static(path.join(__dirname, 'dist', 'assets'), {
  maxAge: '1y',
  immutable: true
}));

// Cache images
app.use('/images', express.static(path.join(__dirname, 'dist', 'images'), {
  maxAge: '30d'
}));

// Serve other static files with short cache
app.use(express.static(path.join(__dirname, 'dist'), {
  maxAge: '1h',
  index: false
}));

// SPA fallback — ALL routes serve index.html
// This is critical for React Router to work on /about, /services, /blog, etc.
app.get('*', (req, res) => {
  res.sendFile(path.join(__dirname, 'dist', 'index.html'));
});

app.listen(PORT, '0.0.0.0', () => {
  console.log(`✅ SYSmoAI production server running on port ${PORT}`);
  console.log(`📁 Serving from: ${path.join(__dirname, 'dist')}`);
});
